
CREATE DATABASE IF NOT EXISTS Pipas01;
USE Pipas01;

CREATE TABLE Pipa (
    Id_pipa INT PRIMARY KEY,
    Modelo_pipa VARCHAR(100),
    Preco DECIMAL(10,2)
);


CREATE TABLE Linha (
    Id_linha INT PRIMARY KEY,
    Tipo_linha VARCHAR(100),
    Preco DECIMAL(10,2)
);


CREATE TABLE Saidas (
    Id_saida INT PRIMARY KEY,
    Id_pipa INT,
    Quantidade INT,
    Data DATE,
    Valor_total DECIMAL(10,2)
);


CREATE TABLE Estoque (
    Id_estoque INT PRIMARY KEY,
    Id_pipa INT,
    Quantidade_estoque INT
);

SELECT 
    Saidas.Data,
    Pipa.Modelo_pipa,
    SUM(Saidas.Quantidade) AS Total_vendido
FROM Saidas
JOIN Pipa ON Saidas.Id_pipa = Pipa.Id_pipa
GROUP BY Saidas.Data, Pipa.Modelo_pipa
ORDER BY Saidas.Data ASC;
